#[derive(Debug, Deserialize)]
pub struct Manifest {
    pub version: u32,
    pub project: String,
    pub modes: HashMap<String, Mode>,
    pub workflows: HashMap<String, Workflow>,
    pub skills_dir: String,
    pub policies_dir: String,
    pub output_dir: String,
}

#[derive(Debug, Deserialize)]
pub struct Mode {
    pub file: String,
    #[serde(default)]
    pub permission: HashMap<String, PermissionRule>,
    #[serde(default)]
    pub policies: Vec<String>,
    #[serde(default)]
    pub skills: Vec<String>,
}

/// A permission entry is either a flat "allow"/"deny"/"ask"
/// or a glob -> action map, e.g. { "src/**": "allow", "*": "deny" }
#[derive(Debug, Deserialize)]
#[serde(untagged)]
pub enum PermissionRule {
    Flat(String),
    Globs(HashMap<String, String>),
}

#[derive(Debug, Deserialize)]
pub struct Workflow {
    pub file: String,
    pub agent: String,
}

pub fn load(path: &str) -> anyhow::Result<Manifest> {
    let text = std::fs::read_to_string(path)
        .map_err(|e| anyhow::anyhow!("reading {path}: {e}"))?;
    let manifest: Manifest = serde_yaml::from_str(&text)
        .map_err(|e| anyhow::anyhow!("parsing {path}: {e}"))?;
    Ok(manifest)
}

Step 3 — src/validate.rs

This is where the cross-reference checks live — catch mistakes before Kilo Code silently ignores a malformed file:
rust

use crate::manifest::Manifest;
use anyhow::{bail, Context};
use std::path::Path;

pub fn run(m: &Manifest) -> anyhow::Result<()> {
    for (name, mode) in &m.modes {
        if !Path::new(&mode.file).exists() {
            bail!("mode '{name}' references missing file: {}", mode.file);
        }
        for policy in &mode.policies {
            if !Path::new(policy).exists() {
                bail!("mode '{name}' references missing policy: {policy}");
            }
        }
        for skill in &mode.skills {
            let skill_dir = Path::new(&m.skills_dir).join(skill);
            let skill_md = skill_dir.join("SKILL.md");
            if !skill_md.exists() {
                bail!("mode '{name}' references missing skill: {skill}");
            }
            check_skill_name(&skill_md, skill)
                .with_context(|| format!("validating skill '{skill}'"))?;
        }
    }

    for (name, wf) in &m.workflows {
        if !Path::new(&wf.file).exists() {
            bail!("workflow '{name}' references missing file: {}", wf.file);
        }
        if !m.modes.contains_key(&wf.agent) {
            bail!("workflow '{name}' references unknown agent/mode: {}", wf.agent);
        }
    }

    Ok(())
}

/// Kilo Code requires SKILL.md's `name:` frontmatter to equal the
/// parent directory name — catch mismatches at build time, not load time.
fn check_skill_name(skill_md: &Path, expected: &str) -> anyhow::Result<()> {
    let text = std::fs::read_to_string(skill_md)?;
    let frontmatter = text
        .strip_prefix("---\n")
        .and_then(|s| s.split_once("\n---"))
        .map(|(fm, _)| fm)
        .context("SKILL.md missing YAML frontmatter delimiters")?;

    let name_line = frontmatter
        .lines()
        .find(|l| l.starts_with("name:"))
        .context("SKILL.md frontmatter missing 'name:' field")?;

    let actual = name_line.trim_start_matches("name:").trim();
    if actual != expected {
        bail!("SKILL.md name '{actual}' does not match directory '{expected}'");
    }
    Ok(())
}

Step 4 — src/render.rs

Assembles the runtime .kilo/ tree from source content:
rust

use crate::manifest::{Manifest, Mode, PermissionRule};
use anyhow::Result;
use std::fs;
use std::path::Path;

pub fn build(m: &Manifest) -> Result<()> {
    let out = Path::new(&m.output_dir);
    render_agents(m, out)?;
    render_commands(m, out)?;
    render_skills(m, out)?;
    Ok(())
}

fn render_agents(m: &Manifest, out: &Path) -> Result<()> {
    let agents_dir = out.join("agents");
    fs::create_dir_all(&agents_dir)?;

    for (name, mode) in &m.modes {
        let body = fs::read_to_string(&mode.file)?;
        let policy_text = concat_policies(&mode.policies)?;
        let frontmatter = render_permission_frontmatter(mode);

        let full = format!(
            "---\ndescription: {name} mode\nmode: primary\n{frontmatter}---\n\n{body}\n\n## Project Policies\n\n{policy_text}\n"
        );

        fs::write(agents_dir.join(format!("{name}.md")), full)?;
    }
    Ok(())
}

fn render_permission_frontmatter(mode: &Mode) -> String {
    if mode.permission.is_empty() {
        return String::new();
    }
    let mut s = String::from("permission:\n");
    for (tool, rule) in &mode.permission {
        match rule {
            PermissionRule::Flat(v) => s.push_str(&format!("  {tool}: {v}\n")),
            PermissionRule::Globs(map) => {
                s.push_str(&format!("  {tool}:\n"));
                for (glob, action) in map {
                    s.push_str(&format!("    \"{glob}\": \"{action}\"\n"));
                }
            }
        }
    }
    s
}

fn concat_policies(paths: &[String]) -> Result<String> {
    let mut out = String::new();
    for p in paths {
        out.push_str(&fs::read_to_string(p)?);
        out.push('\n');
    }
    Ok(out)
}

fn render_commands(m: &Manifest, out: &Path) -> Result<()> {
    let commands_dir = out.join("commands");
    fs::create_dir_all(&commands_dir)?;

    for (name, wf) in &m.workflows {
        let body = fs::read_to_string(&wf.file)?;
        let full = format!(
            "---\ndescription: {name} workflow\nagent: {}\n---\n\n{body}\n",
            wf.agent
        );
        fs::write(commands_dir.join(format!("{name}.md")), full)?;
    }
    Ok(())
}

fn render_skills(m: &Manifest, out: &Path) -> Result<()> {
    let skills_out = out.join("skills");
    fs::create_dir_all(&skills_out)?;

    for entry in fs::read_dir(&m.skills_dir)? {
        let entry = entry?;
        if !entry.path().is_dir() {
            continue;
        }
        let name = entry.file_name();
        let src = entry.path().join("SKILL.md");
        if !src.exists() {
            continue;
        }
        let dest_dir = skills_out.join(&name);
        fs::create_dir_all(&dest_dir)?;
        fs::copy(&src, dest_dir.join("SKILL.md"))?;
    }
    Ok(())
}

#!/usr/bin/env bash
# tools/check-manifest.sh
